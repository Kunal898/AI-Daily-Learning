# ML Pipeline Day 222: Low-Rank Adaptation (LoRA) for Fine-Tuning LLMs
import numpy as np

def compute_loss(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    eps = 1e-15
    y_pred = np.clip(y_pred, eps, 1 - eps)
    # Binary cross entropy
    return -float(np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred)))

y_t = np.array([1, 0, 1, 1])
y_p = np.array([0.9, 0.1, 0.8, 0.95])
print(f"Loss for Low-Rank Adaptation (LoRA) for Fine-Tuning LLMs: {compute_loss(y_t, y_p):.4f}")
